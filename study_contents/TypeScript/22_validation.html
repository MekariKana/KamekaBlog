class Course {
    @Required
    title: string;
    @Required
    @PositiveNumber
    price: number;

    constructor(t: string, p: number) {
        this.title = t;
        this.price = p;
    }
}

// フォームの要素を取得
const courseForm = document.querySelector('form')!;
// formがsubmitされた時のeventを追加する
courseForm.addEventListener('submit', event => {
    // eventのpreventDefalt(httpリクエストをしない設定)を呼び出す
    event.preventDefault();
    // titleとpriceのオブジェクトを取得し、HTMLInputElementにキャスト
    const titleEl = document.getElementById('title') as HTMLInputElement;
    const priceEl = document.getElementById('price') as HTMLInputElement;

    // 各オブジェクトからvalueプロパティを取得し、変数に格納
    const title = titleEl.value;
    const price = +priceEl.value;

    // Corseのインスタンスを生成
    const createdCourse = new Course(title, price);

    // バリデーションの実行
    if (!valudate(createdCourse)) {
        alert('正しく入力してください。');
        return;
    }
    console.log(createdCourse);
})
バリデーションロジック 

// クラス・プロパティ毎のバリデーション設定を保存
interface ValidatorConfig {
    // キー(クラス名)はindex型で定義
    [prop: string]: {
        // バリデーターのListを保持 例['required', 'positive']
        [validatableProp: string]: string[] 
    }
}

// interface型のオブジェクトを定義
// 最初は空のobjectだが、デコレーターが実行された時に値が入力される
const registeredValidators: ValidatorConfig = {};

// registeredValidatorsに値を設定
function Required(target: any, propName: string) {
    registeredValidators[target.constructor.name] = { // クラス名(例：Course)をキーに保存
        // 新しいオブジェクトで上書きしないよう、すでに登録されている内容を追加
        ...registeredValidators[target.constructor.name],
        // 新しいキー・バリューを追加
        // キーはプロパティ名(例：title)、値はバリデーション名(例：required)
        [propName]: [
            // 登録済みバリデーションを消さないよう、キー(propName)から登録済みバリデーションを取得
            // [target.constructor.name]?：クラス名が存在する時だけ[propName]を取得
            // ?? []：左の結果がnull or undefinedなら[](空の配列)を追加
            ...(registeredValidators[target.constructor.name]?.[propName] ?? []),
            "required",
        ]
    }
}

function PositiveNumber(target: any, propName: string) {
    registeredValidators[target.constructor.name] = {
        ...registeredValidators[target.constructor.name],
        [propName]: [
            ...(registeredValidators[target.constructor.name]?.[propName] ?? []),
            "positive",
        ]
    }
}

// 引数のobjectに必要なバリデーションを判別し、実行する
function valudate(obj: any) {
    // objがregisteredValidatorsのどのクラスのインスタンスか、コンストラクター名を取得
    const objValidatorConfig = registeredValidators[obj.constructor.name];
    // registeredValidatorsにバリデーション設定がない場合、検証するものがないのでtrue
    if(!objValidatorConfig) {
        return true;
    }

    let isValid = true;
    // クラスに登録してあるバリデーターリストを取得
    for (const prop in objValidatorConfig) {
        // 上記で取得したproperty名をキーにし、validation名を取得
        for (const validator of objValidatorConfig[prop]){
            // コンソールでバリデーター内容の確認
            console.log(prop);
            switch (validator) {
                case 'required':
                    // !!obj[prop]：propの値が空じゃなければtrue
                    // 1つでもfalseがあればisValidはfalseになる
                    isValid = isValid && !!obj[prop];
                    // switchを抜ける
                    break;
                case 'positive':
                    // obj[prop] > 0：0以上ならtrue
                    isValid = isValid && obj[prop] > 0;
                    break;
            }
        }
    }
    // バリデーターが1つもない場合はfor文を通らないので、デフォルト返却値を設定
    return isValid;
}

デコレーターのまとめ

デコレーターとは、クラス・メソッド・プロパティなどに後から機能や設定を追加するための関数。 「＠xxxxxx」という形式で対象の上に書き加えることで、対象のクラスやメソッドが作られるタイミングで自動的に定義した関数が呼ばれる。 バリデーターでは@Requiredでプロパティに入力必須というルールを追加、@Autobindではメソッドの動きを変更しthisに呼び出し元のオブジェクトを固定するなど、好きな関数を追加できる。 デコレーターを使用することで、同じ処理を何度も書く必要はなくなる。
